📹 TP-Link 监控快捷入口

下一步等你的界面跑通后，我教你如何配置与你手机上 TP-Link App 的直连方式。

顺便给你配套的 Google Apps Script（支持“拉取 + 卖出减/进货加同步”） 如果你之前的 Apps Script 只有接收功能，可以顺便把你的 Sheet 里的 Apps Script 换成下面这段，这样 「从 Sheet 拉取最新货单」 和 「卖出减/进货加同步」 就都能双向打通（表格第 1 列放货品名，第 2 列放数量）： JavaScript function doGet() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const rows = sheet.getDataRange().getValues(); const items = []; for (let i = 1; i < rows.length; i++) { if (rows[i][0]) { items.push({ name: rows[i][0], qty: Number(rows[i][1]) || 0 }); } } return ContentService.createTextOutput(JSON.stringify({ items })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const payload = JSON.parse(e.postData.contents); const items = payload.items || []; sheet.clearContents(); sheet.appendRow(["货品名称", "当前库存", "最后更新时间", "变动备注"]); items.forEach(item => { sheet.appendRow([item.name, item.qty, payload.updatedAt || new Date(), payload.note || ""]); }); return ContentService.createTextOutput("OK"); }
